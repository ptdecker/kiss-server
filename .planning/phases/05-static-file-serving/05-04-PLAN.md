<objective>
Fix the inconsistent 404 body format between the router's built-in NotFoundHandler and the handlers module not_found() helper.

Purpose: UAT test 6 revealed that path traversal rejections (which route through NotFoundHandler in router.rs) produce a body of "Not Found" without a trailing newline, while all other 404s from StaticFileHandler use the not_found() helper which produces "Not Found\n". This inconsistency surfaced as a minor UAT failure.

Output: src/server/router.rs with NotFoundHandler body changed from b"Not Found" to b"Not Found\n", Content-Length updated from "9" to "10", and a test verifying the body format.
</objective>

fn dispatch_unmatched_body_has_trailing_newline() {
    // NotFoundHandler body must match not_found() helper: "Not Found\n" (with newline)
    // Regression test for UAT test 6: path traversal returned 404 without trailing newline
    let router = Router::new();
    let mut ctx = make_ctx(RequestMethod::Get, "/missing");
    router.dispatch(&mut ctx).unwrap();
    let mut buf: Vec<u8> = Vec::new();
    ctx.response.write_to(&mut buf).unwrap();
    let output = String::from_utf8(buf).unwrap();
    let sep_pos = output.find("\r\n\r\n").expect("no blank separator in 404 response");
    let body = &output[sep_pos + 4..];
    assert_eq!(
        body, "Not Found\n",
        "NotFoundHandler body must end with newline, got: {:?}",
        body
    );
}
